CREATE TABLE incident (
    incident_id SERIAL PRIMARY KEY,
    incident_number VARCHAR(30),
    title VARCHAR(200),

    affected_users INTEGER,
    application VARCHAR(100),
    root_cause VARCHAR(200),
    created_by INTEGER REFERENCES app_user(user_id),
